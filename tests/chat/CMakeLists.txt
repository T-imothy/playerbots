# Source-derived chat policy checks shared by all CMaNGOS variants.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
foreach(_kind admission audience)
  set(_out "${CMAKE_CURRENT_BINARY_DIR}/chat_${_kind}")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${playerbots_SOURCE_DIR}/tests/chat/run_chat_${_kind}_test.py"
    --generate-only --output "${_out}" RESULT_VARIABLE _generated)
  if(NOT _generated EQUAL 0)
    message(FATAL_ERROR "Could not generate native chat ${_kind} regression")
  endif()
  add_executable(arch4_chat_${_kind}_tests "${_out}/test.cpp")
endforeach()
add_executable(arch4_chat_prefix_tests "${playerbots_SOURCE_DIR}/tests/chat/ChatPrefixTest.cpp")
add_executable(arch4_chat_sender_tests "${playerbots_SOURCE_DIR}/tests/chat/ChatBroadcastSenderTest.cpp")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${playerbots_SOURCE_DIR}/tests/chat/generate_packet_test.py"
  --core "${PROJECT_SOURCE_DIR}" --output "${CMAKE_CURRENT_BINARY_DIR}/chat_packet.cpp" RESULT_VARIABLE _generated)
if(NOT _generated EQUAL 0)
  message(FATAL_ERROR "Could not generate native chat packet regression")
endif()
add_executable(arch4_chat_packet_tests "${CMAKE_CURRENT_BINARY_DIR}/chat_packet.cpp")
if(CMAKE_PROJECT_NAME MATCHES "Classic")
  target_compile_definitions(arch4_chat_packet_tests PRIVATE MANGOSBOT_ZERO)
elseif(CMAKE_PROJECT_NAME MATCHES "TBC")
  target_compile_definitions(arch4_chat_packet_tests PRIVATE MANGOSBOT_ONE)
elseif(CMAKE_PROJECT_NAME MATCHES "WoTLK")
  target_compile_definitions(arch4_chat_packet_tests PRIVATE MANGOSBOT_TWO)
endif()
foreach(_kind admission audience prefix sender packet)
  target_compile_features(arch4_chat_${_kind}_tests PRIVATE cxx_std_20)
  target_include_directories(arch4_chat_${_kind}_tests PRIVATE "${playerbots_SOURCE_DIR}")
  target_link_libraries(arch4_chat_${_kind}_tests PRIVATE cmangos-compile-option-interface Boost::headers)
  if(MSVC)
    target_compile_options(arch4_chat_${_kind}_tests PRIVATE /UNDEBUG)
  else()
    target_compile_options(arch4_chat_${_kind}_tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME arch4_chat_${_kind} COMMAND arch4_chat_${_kind}_tests)
endforeach()
